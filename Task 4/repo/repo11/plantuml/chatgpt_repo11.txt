  }

  package "Forwarder Proxies" as Proxies {
    component "MatryxForwarder\n(Generic Call Forwarder)\n(Finds MatryxPlatform by version/system,\nForwards calldata + injected params)" as Forwarder <<contract>>

    component "MatryxTournament\n(Proxy Facade)\n(Forwards tournament-related calls)" as TournamentProxy <<contract>>
  }

  package "Library Implementations\n(delegatecall targets)" as Libs {
    component "LibPlatform\n(Platform API:\ninfo/query/admin,\ncreateTournament)" as LibPlatform <<library>>

    component "LibTournament\n(Tournament API:\nround lifecycle,\nentries/submissions,\nwinners,\nwithdraw/recover)" as LibTournament <<library>>

' --- Primary user interactions
User --> CommitProxy : commit-related calls
User --> TournamentProxy : tournament-related calls
User --> Platform : platform API (direct)

Owner --> System : manage versions/contracts/types\n(createVersion/setVersion/setContract/\nsetLibraryName/addContractMethods)
Owner --> Platform : admin actions\n(blacklist, token upgrade,\nwithdraw tokens, ownership transfer)

' --- Forwarding / dispatch flow
CommitProxy -down-|> Forwarder
TournamentProxy -down-|> Forwarder

Forwarder --> System : getContract(version,"MatryxPlatform")
Forwarder --> Platform : forward calldata\n(+ caller + version injection)

Platform --> System : getVersion()/getLibraryName(msg.sender)\ngetContract(version, libName)\ngetContractMethod(version,cName,selector)
Platform --> LibPlatform : delegatecall (selected)
Platform --> LibCommit : delegatecall (selected)
Platform --> LibTournament : delegatecall (selected)

' --- Domain-level dependencies
LibPlatform --> System : getVersion(), setContractType()\n(register new tournament proxy type)
LibPlatform --> TournamentProxy : deploy new MatryxTournament(version, system)
LibPlatform --> MatryxToken : transferFrom()\n(bounty funding)
LibCommit --> MatryxToken : transferFrom()/transfer()\n(commit staking/rewards)
LibTournament --> MatryxToken : transferFrom()/transfer()\n(entry fees, rewards, withdrawals)

' --- Shared utilities
LibPlatform ..> LibGlobals
LibCommit ..> LibGlobals
LibTournament ..> LibGlobals
LibPlatform ..> SafeMath57
LibCommit ..> SafeMath57
LibTournament ..> SafeMath57

' --- Token internal structure (coarse)
MatryxToken ..> ERC20Impl
MatryxToken ..> Upgradeable
MatryxToken ..> TokenControls
ERC20Impl ..> Ownable50
TokenControls ..> Ownable50
Upgradeable ..> Ownable50

' --- Ownership
System ..> Ownable57
Platform ..> SafeMath57

' --- Off-chain toolchain relationships
Migrations --> Ganache : deploy
Migrations --> System
Migrations --> Platform
Migrations --> CommitProxy
Migrations --> MatryxToken
Migrations --> LibPlatform
Migrations --> LibCommit
Migrations --> LibTournament

SetupGen --> System : generate & execute\naddContractMethods/addContractMethod(s)
GasParser --> Ganache : watch transactions\n(JsonRpcProvider)
Utils --> Ganache : client interactions\n(ethers/web3 wrappers)
Utils --> Platform : invoke flows\n(setup/bench/test)
