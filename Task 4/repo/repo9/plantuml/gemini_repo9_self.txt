    component "extract.js\n(Node.js)" as Extractor
    note right of Extractor

        CODE:BEGIN / CODE:END blocks
    end note

    component "graphql.js\n(Node.js)" as SchemaParser
    note right of SchemaParser
        Transforms GraphQL SDL
        to JSON representation
    end note

    component "render.sh\n(Shell)" as Renderer
}

package "Generated Data" {
    file "data/examples.json" as ExamplesData
    file "data/eos/graphql.json" as SchemaData
    file "Static Images\n(PNG)" as Images
}

    component "Hugo\n(Static Site Generator)" as Hugo
    component "Netlify\n(CI/CD)" as Netlify
}

' Extraction Flow

ExternalSchemas --> SchemaParser : "Input"

' Diagram Rendering

' Site Generation
