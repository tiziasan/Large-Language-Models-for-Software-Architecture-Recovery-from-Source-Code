    + String getTitolo()

    + boolean equals(Object obj)
    + String toString()

  }

  class Cartaceo {
    + int pagine

    + Cartaceo(int pagine, int num_autori, String titolo, Autore a) throws IllegalArgumentException
    + String toString()
  }

  class Digitale {
    + int kByte

    + Digitale(int kByte, int num_autori, String titolo, Autore a) throws IllegalArgumentException
    + String toString()
  }

  class GestoreArrticoli {
    + {static} void main(String[] args)

    + MiaEccezioneListaPiena()
  }

  '========================
  ' Inheritance
  '========================
  Libro <|-- Cartaceo
  Libro <|-- Digitale
  Exception <|-- MiaEccezioneLibroEsistente
  Exception <|-- MiaEccezioneListaPiena

  '========================
  ' Associations (from fields)
  '========================
  ArticoliLettura "1" --> "0..3" Libro : articoli

  Libro "1" --> "0..*" Autore : autori
